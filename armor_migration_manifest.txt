## 八、迁移脚本示例

可以使用以下 PowerShell 脚本进行批量重命名（示例，需要根据实际情况调整）:

```powershell
# 设置基础路径
$basePath = "Z:\Git\GregTech-Construct\src\main\resources\assets\tconstruct"

# 定义替换规则
$rules = @{
    "armor_plate" = "layer_plate"
    "armor_mail" = "layer_mail"
    "cast_boots" = "core_boots"
    "cast_chestplate" = "core_chestplate"
    "cast_helmet" = "core_helmet"
    "cast_leggings" = "core_leggings"
    "frame_of_boots" = "frame_boots"
    "frame_of_chestplate" = "frame_chestplate"
    "frame_of_helmet" = "frame_helmet"
    "frame_of_leggings" = "frame_leggings"
    "massive_cast_boots" = "heavy_core_boots"
    "massive_cast_chestplate" = "heavy_core_chestplate"
    "massive_cast_helmet" = "heavy_core_helmet"
    "massive_cast_leggings" = "heavy_core_leggings"
}

# 批量重命名函数
function Rename-ArmorFiles {
    param($directory)
    Get-ChildItem -Path $directory -Recurse -File | ForEach-Object {
        $newName = $_.Name
        foreach ($old in $rules.Keys) {
            if ($_.Name -match $old) {
                $newName = $newName -replace $old, $rules[$old]
                Rename-Item -Path $_.FullName -NewName $newName -WhatIf
                break
            }
        }
    }
}
```

**注意**: 上述脚本使用了 `-WhatIf` 参数，仅显示将要执行的操作而不实际执行。验证无误后，移除 `-WhatIf` 参数以实际执行重命名。
